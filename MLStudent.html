oportunitati=[]
nr_max_sfaturi=3
for i, coloana in enumerate(X_train.columns):
    panta=model.coef_[i]
    if panta > 0:
        valoare_curenta=Student[coloana].values[0]
        max=limite_maxime[coloana]
        if valoare_curenta<max:
            spatiu_crestere=max-valoare_curenta
            castig_potential=spatiu_crestere*panta
            if castig_potential>0.1:
                oportunitati.append({
                    'Factor':coloana,
                    'Situatie_actuala':valoare_curenta,
                    'Tinta':max,
                    'Crestere_Nota':castig_potential
                })
df_sfaturi=pd.DataFrame(oportunitati)
print(f"---Raport imbunatatire---")
if not df_sfaturi.empty:
    df_sfaturi=df_sfaturi.sort_values(by='Crestere_Nota',ascending=False)
    sfaturi_fin=df_sfaturi.head(nr_max_sfaturi)
    contor=1
    for index,rand in sfaturi_fin.iterrows():
        print(f"\nSFAT {contor}:Trebuie sa mai imbunatatesti '{rand['Factor']}'")
        print(f"\nImpact:Nota ta ar creste cu +{rand['Crestere_Nota']:.2f} puncte")
        contor += 1
    castig_total=sfaturi_fin['Crestere_Nota'].sum()
    print(f"Daca respecti aceste {len(sfaturi_fin)} sfaturi, nota creste cu aprox. {castig_total:.2f} puncte!")
